import { getDatabase, ref, get, child } from "firebase/database"; // npm modüler import async function fetchBinalar() { const container = document.getElementById('bina-listesi'); container.innerHTML = ''; // 🔹 Yükleniyor mesajı const loadingMsg = document.createElement('div'); loadingMsg.textContent = "Veriler yükleniyor..."; loadingMsg.style.textAlign = "center"; loadingMsg.style.fontSize = "1.2rem"; loadingMsg.style.color = "#333"; container.appendChild(loadingMsg); try { const snapshot = await get(child(ref(getDatabase()), 'binalar')); console.log("Firebase snapshot:", snapshot.val()); // 🔹 Log eklendi container.innerHTML = ''; // loading mesajını kaldır const data = snapshot.val(); if (!data) { const msg = document.createElement('div'); msg.textContent = "Hiç bina bulunamadı."; msg.style.textAlign = "center"; msg.style.fontSize = "1.2rem"; msg.style.color = "#777"; container.appendChild(msg); return; } // Veri varsa kartları oluştur Object.keys(data).forEach(bina => { const count = Number(data[bina].sayac) || 0; const card = document.createElement('div'); card.className = 'bina-card'; card.innerHTML = `
${bina}
${count}
`; container.appendChild(card); }); } catch (error) { container.innerHTML = ''; const msg = document.createElement('div'); msg.textContent = "Veriler alınamadı. Lütfen daha sonra tekrar deneyin."; msg.style.textAlign = "center"; msg.style.fontSize = "1.2rem"; msg.style.color = "#f44336"; container.appendChild(msg); console.error("Firebase hatası:", error); // 🔹 Hata logu } }
